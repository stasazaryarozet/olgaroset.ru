## Шаг 3: Синхронизация seats.json

### Создать скрипт синхронизации

**Файл:** `olga/olgaroset.ru/chl/sync_seats.py`

```python
#!/usr/bin/env python3
"""
Синхронизация seats.json из Google Sheet
Обновляет статический JSON файл для fallback
"""

import json
import gspread
from google.oauth2.service_account import Credentials
from datetime import datetime

# Конфигурация
SHEET_ID = '1V63W-BMWKqM9M4IabRJch4rdCVNL3DMGBugDzOxdeTQ'  # Или из CONFIG
SHEET_NAME = 'ЦДЛ 2 декабря'
CELL_RANGE = 'G2'  # Ячейка "Свободно"
TOTAL_SEATS = 25

def sync_seats():
    """Синхронизировать seats.json из Google Sheet"""
    try:
        # Подключение к Google Sheets
        # Требуется credentials.json (service account)
        scope = ['https://spreadsheets.google.com/feeds',
                 'https://www.googleapis.com/auth/drive']
        creds = Credentials.from_service_account_file('credentials.json', scopes=scope)
        client = gspread.authorize(creds)
        
        # Открыть Sheet
        sheet = client.open_by_key(SHEET_ID)
        worksheet = sheet.worksheet(SHEET_NAME)
        
        # Получить значение из G2
        seats_left = int(worksheet.acell(CELL_RANGE).value)
        
        # Обновить seats.json
        data = {
            "seatsLeft": seats_left,
            "totalSeats": TOTAL_SEATS,
            "lastUpdated": datetime.utcnow().isoformat() + "Z",
            "status": "available" if seats_left > 0 else "sold_out"
        }
        
        with open('seats.json', 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        
        print(f"✅ Обновлено: {seats_left} мест свободно")
        return True
        
    except Exception as e:
        print(f"❌ Ошибка синхронизации: {e}")
        return False

if __name__ == '__main__':
    sync_seats()
```

**Альтернатива:** Использовать Google Sheets API через публичный доступ (если Sheet публичный).
